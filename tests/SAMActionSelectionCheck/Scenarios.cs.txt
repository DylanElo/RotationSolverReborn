public static int Run(){int count=0,failed=0;void Check(string name,Func<bool> check){count++;try{if(check())Console.WriteLine("PASS "+name);else{failed++;Console.WriteLine("FAIL "+name);}}catch(Exception e){failed++;Console.WriteLine("FAIL "+name+" "+e.GetBaseException().Message);}}
