class FakeAction : IBaseAction {
 public ActionID ID {get;} public FakeCooldown Cooldown {get;}=new(); public FakeTarget Target {get;}=new();
 public FakeInfo Info=new(); public FakeSetting Setting {get;set;}=new(); public FakeConfig Config {get;set;}=new();public FakeTargetInfo TargetInfo=new();public bool EnoughLevel=true,IsEnabled=true,Available,ThrowOnUse;
 public Func<bool,bool>? StatusCheck;
 public IBattleChara[]? NativeCandidates;
 public FakeAction(ActionID id){ID=id;Info.Eligibility=()=>IsEnabled&&EnoughLevel;Setting.TargetStatusProvide=id switch {ActionID.IronJawsPvE=>[StatusID.Stormbite,StatusID.Windbite,StatusID.CausticBite,StatusID.VenomousBite],ActionID.StormbitePvE=>[StatusID.Stormbite],ActionID.CausticBitePvE=>[StatusID.CausticBite],ActionID.WindbitePvE=>[StatusID.Windbite],ActionID.VenomousBitePvE=>[StatusID.VenomousBite],_=>null};}
 public bool CanUse(out IAction? act,bool usedUp=false,bool skipTTKCheck=false,bool skipComboCheck=false,bool skipStatusProvideCheck=false,bool skipAoeCheck=false) {
  if(ThrowOnUse)throw new InvalidOperationException("simulated native failure");
  var ok=Available&&IsEnabled&&EnoughLevel&&Info.EnoughLevelAndQuest()&&Cooldown.WillHaveOneCharge(DataCenter.DefaultGCDRemain)
   && (skipTTKCheck||DataCenter.AverageTTK>=Config.TimeToKill)
   && (Setting.StatusNeed==null||!StatusHelper.PlayerWillStatusEnd(Config.StatusRefreshGcdCount*2.5f,true,Setting.StatusNeed))
   && (skipStatusProvideCheck||Setting.StatusProvide==null||StatusHelper.PlayerWillStatusEnd(Config.StatusRefreshGcdCount*2.5f,true,Setting.StatusProvide))
   && (StatusCheck?.Invoke(skipStatusProvideCheck)??true);
  if(ok&&NativeCandidates!=null){var target=NativeCandidates.FirstOrDefault(t=>Setting.CanTarget(t)&&(skipStatusProvideCheck||Setting.TargetStatusProvide==null||t.WillStatusEndGCD(Config.StatusRefreshGcdCount,0,true,Setting.TargetStatusProvide)));ok=target!=null;if(ok)Target.Target=target;}
  else if(ok&&Target.Target!=null)ok=Setting.CanTarget(Target.Target);
  act=ok?this:null; return ok;
 }
}
