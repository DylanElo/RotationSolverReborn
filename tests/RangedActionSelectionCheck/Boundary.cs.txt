 public Func<bool,bool>? StatusCheck;

 public bool CanUse(out IAction? act,bool usedUp=false,bool skipTTKCheck=false,bool skipComboCheck=false,bool skipStatusProvideCheck=false) {
  var ok=Available&&IsEnabled&&EnoughLevel&&Info.EnoughLevelAndQuest()&&Cooldown.WillHaveOneCharge(DataCenter.DefaultGCDRemain)
   && (skipTTKCheck||DataCenter.AverageTTK>=Config.TimeToKill)
   && (StatusCheck?.Invoke(skipStatusProvideCheck)??true); act=ok?this:null; return ok;
 }
}
