 static bool FitsWeave(float now,float nextGcd,float currentGcdTotal)=>nextGcd-now>Math.Max(.5f,currentGcdTotal*.25f);
 static float RetryWeaveAt(float now)=>now+PollStep;
 static bool HasWeaveBudget(int confirmed)=>confirmed<2;
 static void CheckLateWeavePolling(){
  bool Select(float ready,float next){for(float at=.72f;FitsWeave(at,next,2.1f);at=RetryWeaveAt(at))if(at>=ready)return true;return false;}
  if(!HasWeaveBudget(0)||!HasWeaveBudget(1)||HasWeaveBudget(2))throw new Exception("Shared two-weave confirmed-effect budget failed");
  if(!Select(.80f,1.5f))throw new Exception("Late-ready ability was skipped after an empty weave poll");
  if(Select(1.0f,1.5f)||FitsWeave(0,.525f,2.1f))throw new Exception("Ability admitted inside native action-ahead/command cutoff");
 }
 public static int Run(string[] args){
  CheckLateWeavePolling();
  string Arg(string key,string fallback){var i=Array.IndexOf(args,key);return i<0?fallback:args[i+1];}
  var fixtures=Arg("--fixtures","");var output=Arg("--output","");Directory.CreateDirectory(output);
  var summaries=new List<object>();

   var counts=engine.Rows.GroupBy(x=>x.name).ToDictionary(g=>g.Key,g=>g.Count());
   var reference=f.reference.GroupBy(x=>x.name).ToDictionary(g=>g.Key,g=>g.Count());
   var packageStats=new{freeReawakens=engine.Rows.Count(r=>r.name=="Reawaken"&&r.free),paidReawakens=engine.Rows.Count(r=>r.name=="Reawaken"&&!r.free),reawakenStartsUnderPotion=engine.Rows.Count(r=>r.name=="Reawaken"&&r.medicated),ouroborosUnderPotion=engine.Rows.Count(r=>r.name=="Ouroboros"&&r.medicated),ouroborosUnderPartyBuff=engine.Rows.Count(r=>r.name=="Ouroboros"&&r.buffs>0),furyUnderPotion=engine.Rows.Count(r=>r.name=="Uncoiled Fury"&&r.medicated),furyUnderPartyBuff=engine.Rows.Count(r=>r.name=="Uncoiled Fury"&&r.buffs>0)};
   var summary=new{packageStats,f.label,f.duration,f.events,f.pages,counts,reference,gauge=engine.Gauge,coils=engine.Coils,free=engine.Free,overcapGauge=engine.GaugeLost,overcapCoils=engine.CoilsLost,unfinished=engine.Gen>0,engine.CombosExpired,engine.TokenExpired,engine.FollowupsExpired,engine.DreadExpired,engine.GenerationExpired,engine.GcdIdleSeconds,engine.Illegal,model="Own state and production GeneralGCD/EmergencyAbility/AttackAbility/hooks/scheduler; observed party buff, potion, targetability and target death windows. No damage/rDPS model; known future boss boundaries emulate timeline support."};
   summaries.Add(summary);File.WriteAllText(Path.Combine(output,f.label+"-timeline.json"),JsonSerializer.Serialize(engine.Rows,new JsonSerializerOptions{WriteIndented=true}));
   Console.WriteLine(f.label+" RA/Ouro/Fury/Ire/Vice "+string.Join("/",new[]{"Reawaken","Ouroboros","Uncoiled Fury","Serpent's Ire","Vicewinder"}.Select(n=>counts.GetValueOrDefault(n)))+" reference "+string.Join("/",new[]{"Reawaken","Ouroboros","Uncoiled Fury","Serpent's Ire","Vicewinder"}.Select(n=>reference.GetValueOrDefault(n)))+" illegal="+engine.Illegal);
   if(engine.GcdIdleSeconds>2)throw new Exception("Unexplained active-target GCD stall "+f.label+" "+engine.GcdIdleSeconds);
   var sequence=false;foreach(var row in engine.Rows){if(row.name=="Reawaken"){if(sequence)throw new Exception("Reawaken restarted before Ouroboros "+f.label);sequence=true;}if(row.name=="Ouroboros"){if(!sequence)throw new Exception("Ouroboros without Reawaken");sequence=false;}}if(sequence)throw new Exception("Unfinished Reawaken sequence "+f.label);
   if(engine.Illegal!=0)throw new Exception("Illegal modeled cast "+f.label);
  }
  File.WriteAllText(Path.Combine(output,"summary.json"),JsonSerializer.Serialize(summaries,new JsonSerializerOptions{WriteIndented=true}));return 0;
 }
 class Engine {

    if(HasWeaveBudget(weavesUsed)&&now>=nextWeave&&(FitsWeave(now,nextGcd,currentGcdTotal)||pending.Count>0&&now>=nextGcd)&&Boundary.CurrentTarget!=null){Update();predicted=s.GCD();var next=predicted??s.Potion;var a=s.Weave(next)??s.Attack(next);if(a!=null){CastAction(a.ID,true);nextWeave=now+.72f;nextGcd=Math.Max(nextGcd,now+ExecutionLock);}else nextWeave=RetryWeaveAt(now);}
   }}

   var selected=f.switches.LastOrDefault(x=>x.t<=now);var target=selected==null?null:targets[selected.target];var targetMeta=target==null?null:f.targets.First(x=>(ulong)x.id==target.GameObjectId);
   if(target!=null){var change=f.targetability.LastOrDefault(x=>x.target==(int)target.GameObjectId&&x.t<=now);if((change!=null&&!change.on)||now>targetMeta!.end)target=null;}
   Boundary.CurrentTarget=Boundary.HostileTarget=target;
   foreach(var b in targets.Values){var meta=f.targets.First(x=>(ulong)x.id==b.GameObjectId);b.TTK=Math.Max(0,meta.end-now);b.Dying=b.TTK<=10f||f.hpDying.LastOrDefault(x=>(ulong)x.target==b.GameObjectId&&x.t<=now)?.dying==true;b.Statuses.Clear();}

   s.WeaponRemain=Math.Max(0,nextGcd-now);s.EncounterTime=now;s.PartyBuffsVersion++;s._decisionComputedAt=float.MinValue;s.Downtime=Math.Min(downtime,target?.TTK??float.MaxValue);s.ReplayBmrDowntime=downtime;s.BMRDowntimeEndIn=float.MaxValue;s.BMRVulnerableIn=float.MaxValue;s.BMRVulnerableEndIn=float.MaxValue;
